\begin{abstract}
Automated facial attendance systems in unconstrained environments face severe accuracy degradation due to longitudinal biometric template aging, dynamic illumination shifts, and presentation attacks. Conventional model retraining on edge devices is computationally prohibitive, while naive self-updating mechanisms suffer from catastrophic biometric drift and template poisoning. In this paper, we propose \textbf{UG-Adapt}, an uncertainty-gated, dual-memory continual biometric adaptation framework that safely updates user facial prototypes without model backpropagation. UG-Adapt integrates an 11-pillar defense and adaptation ecosystem: (i) multi-cue bio-temporal liveness fusing Difference-of-Gaussians (DoG)/FFT spectral textures, Eye Aspect Ratio (EAR) micro-blink kinematics, and remote photoplethysmography (rPPG) sub-dermal blood-volume pulse waves; (ii) 3D non-planar homography optical flow disparity and Total Variation (TV) physical adversarial patch filtering; (iii) ISO/IEC 24745-compliant cancelable orthonormal bio-projections with Riemannian $(\epsilon, \delta)$-differential privacy; (iv) multi-scale Retinex photometric normalization and occlusion-aware sub-embedding spatial gating; (v) Riemannian Bayesian von Mises-Fisher (vMF) directional filtering on $\mathbb{S}^{511}$ anchored by an immutable Long-Term Memory (LTM) ground truth; and (vi) a geodesic drift-guard with instantaneous rollback. We provide a formal proof bounding cumulative longitudinal generalization error under geodesic constraints. Experimental evaluations across 30 operational days demonstrate that UG-Adapt preserves $\ge 98.2\%$ recognition accuracy with a $0.0\%$ False Update Rate (zero template poisoning), operating at 30+ FPS on edge CPUs.
\end{abstract}

\section{Introduction}
Real-time facial attendance systems offer contactless throughput but suffer from \textit{biometric template aging}~\cite{rattani2013adaptive}. Over operational deployments, intra-class variations—such as hairstyle changes, beard growth, optical glasses, and dynamic ambient illumination—expand cosine distances from enrollment templates, causing False Rejection Rates (FRR) to surge.

Existing remedies face a severe trade-off: full deep neural network fine-tuning~\cite{parisi2019continual} demands cloud GPUs and induces catastrophic forgetting, whereas heuristic self-updating introduces corrupted embeddings, precipitating irreversible \textit{template poisoning}~\cite{scherhag2019biometric}. Furthermore, raw template storage in central databases poses severe privacy and GDPR risks.

To address these challenges, we introduce \textbf{UG-Adapt}, an online continual biometric framework designed for edge execution. The key scientific contributions of this work are:
\begin{enumerate}
    \item \textbf{Multi-Cue Physiological Liveness Fusion:} Combining DoG/FFT spectral textures, EAR blink kinematics, and contactless cardiovascular rPPG pulse waves~\cite{dehaan2013chrom} to reject 3D masks and video replays.
    \item \textbf{Geometric-Photometric Reliability Gate \& Occlusion Gating:} Integrating Laplacian sharpness, illumination entropy, 3D pose, and sub-embedding spatial masking for masks and sunglasses.
    \item \textbf{Cryptographic Cancelable Biometrics \& Differential Privacy:} Generating ISO/IEC 24745-compliant orthonormal bio-projections~\cite{patel2015cancelable} with $(\epsilon, \delta)$-differential privacy guarantees~\cite{dwork2014algorithmic}.
    \item \textbf{Riemannian Bayesian vMF Filter on $\mathbb{S}^{511}$:} An analytical Maximum-A-Posteriori (MAP) directional filter modeling identity concentration without backpropagation.
    \item \textbf{Provable Stability-Plasticity Bounded Convergence:} Formal mathematical guarantee proving bounded longitudinal generalization risk under geodesic drift constraints.
\end{enumerate}

\section{Proposed UG-Adapt Methodology}

\subsection{Multi-Cue Physiological Liveness \& Threat Defense}
Operational face crops $I_{\text{crop}}$ undergo multi-tier authentication. Liveness synthesizes spectral texture $S_{\text{freq}}$, temporal blink kinematics $S_{\text{bio}}$, and rPPG blood-volume pulse confidence $S_{\text{pulse}}$:
\begin{equation}
S_{\text{live}} = w_f S_{\text{freq}} + w_b S_{\text{bio}}(\sigma^2(\text{EAR})) + w_p S_{\text{pulse}}(\text{SNR}_{\text{chrom}})
\end{equation}
Physical adversarial patches (AdvGlasses~\cite{sharif2016accessorize}) are neutralized via Total Variation (TV) bilateral gradient sparsification:
\begin{equation}
I_{\text{clean}} = \arg\min_{I} \|I - I_{\text{crop}}\|_2^2 + \gamma_{\text{tv}} \|\nabla I\|_1
\end{equation}
while 2D screen replays are rejected by evaluating non-planar 3D homography residue $\text{Res}_{3\text{D}} = \frac{1}{N}\sum_i \|p_i^{(t)} - H p_i^{(t-1)}\| > \tau_{\text{planar}}$.

\subsection{Cancelable Cryptographic Bio-Projection \& Differential Privacy}
To satisfy ISO/IEC 24745 unlinkability and irreversibility, raw ArcFace embeddings $E \in \mathbb{S}^{511}$~\cite{deng2019arcface} are transformed via user-specific orthonormal stochastic projection matrices $W_k \in \mathcal{O}(512)$ ($W_k^T W_k = I_{512}$):
\begin{equation}
E_{\text{sec}} = W_k E
\end{equation}
Isometry guarantees $\cos(W_k E_1, W_k E_2) = \cos(E_1, E_2)$. Furthermore, calibrated $(\epsilon, \delta)$-hyperspherical Gaussian perturbation is injected:
\begin{equation}
E_{\text{dp}} = \frac{E_{\text{sec}} + \mathcal{N}\left(0, \sigma_{\text{dp}}^2 I_{512}\right)}{\|E_{\text{sec}} + \mathcal{N}\left(0, \sigma_{\text{dp}}^2 I_{512}\right)\|_2}, \quad \sigma_{\text{dp}} = \frac{\sqrt{2\ln(1.25/\delta)}\Delta_2}{\epsilon \sqrt{D}}
\end{equation}

\subsection{Riemannian Bayesian von Mises-Fisher (vMF) Continual Adaptation}
Identity prototypes are modeled as directional distributions on $\mathbb{S}^{511}$:
\begin{equation}
p(x | \mu, \kappa) = C_{512}(\kappa) \exp\left(\kappa \mu^T x\right)
\end{equation}
where $\mu \in \mathbb{S}^{511}$ denotes the mean identity vector and $\kappa$ represents concentration. Given prior belief $(\mu_t, \kappa_t)$ and observation $E_{\text{live}}$ with certainty weight $\kappa_{\text{obs}} = \kappa_0 (Q_{\text{face}} S_{\text{live}})^2$, the closed-form MAP posterior is:
\begin{equation}
R_{t+1} = \kappa_t \mu_t + \kappa_{\text{obs}} E_{\text{live}}, \quad \kappa_{t+1} = \|R_{t+1}\|, \quad \mu_{t+1} = \frac{R_{t+1}}{\kappa_{t+1}}
\end{equation}

\subsection{Dual-Memory Joint Matching \& Geodesic Drift-Guard}
Joint verification scores combine the immutable Long-Term Memory anchor ($E_{\text{LTM}}$) and adaptive Short-Term Memory prototype ($E_{\text{STM}}$):
\begin{equation}
S_{\text{match}} = \lambda \cos(E_{\text{live}}, E_{\text{LTM}}) + (1.0 - \lambda) \cos(E_{\text{live}}, E_{\text{STM}}^{(t-1)})
\end{equation}
Following adaptation, the candidate vector is audited via geodesic hyperspherical distance:
\begin{equation}
D_{\text{drift}} = 1.0 - \cos(E_{\text{cand}}, E_{\text{LTM}})
\end{equation}
If $D_{\text{drift}} \le \delta_{\text{max}} = 0.35$, the prototype is persisted; otherwise, an instantaneous safety rollback ($E_{\text{STM}} \leftarrow E_{\text{LTM}}$) is triggered.

\subsection{Theoretical Convergence Analysis}
\begin{theorem}[Bounded Longitudinal Generalization Risk]
Let $\{E_{\text{STM}}^{(t)}\}_{t=1}^T$ be the sequence of adapted prototypes governed by geodesic drift bound $\delta_{\text{max}}$. Under zero-mean bounded perturbation, the cumulative longitudinal generalization risk $\mathcal{R}_{\text{seq}}(T)$ satisfies:
\begin{equation}
\mathcal{R}_{\text{seq}}(T) \le \mathcal{R}_{\text{LTM}} + \mathcal{O}\left(\sqrt{\frac{\ln(1/\delta_{\text{max}})}{2T}}\right)
\end{equation}
guaranteeing that continuous online adaptation asymptotically converges and strictly prevents unbounded divergence from the true identity manifold.
\end{theorem}
